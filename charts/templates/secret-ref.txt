# ============================================================
# [大規模] Secret 參考範本
# ============================================================
# 這個檔案「不部署」，僅作為 SP 人員手動建立 Secret 的參考指令
# 機密值不進 git，由 SP / CI secrets 注入
#
# 建立指令（SP 人員執行）：
#   kubectl create secret generic sep-engine-secret \
#     --from-literal=DB_PASSWORD=<your-password> \
#     --from-literal=JWT_SECRET=<your-jwt-secret> \
#     --from-literal=API_KEY=<your-api-key> \
#     -n ns-sep-{ENV}
#
# 或搭配 External Secrets Operator 從 Vault / GCP SM 自動拉：
#   apiVersion: external-secrets.io/v1beta1
#   kind: ExternalSecret
#   metadata:
#     name: sep-engine-secret
#   spec:
#     refreshInterval: 1h
#     secretStoreRef:
#       name: vault-backend
#       kind: SecretStore
#     target:
#       name: sep-engine-secret
#     data:
#       - secretKey: DB_PASSWORD
#         remoteRef:
#           key: sep/engine
#           property: db_password
